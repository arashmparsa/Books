\section{Reading Methodology Notes}

\subsection{Key Learning Principles}
\begin{itemize}
  \item \textbf{Idiosyncratic books with nuance improve recall}
  \item \textbf{Spacing}: Distribute learning over time
  \item \textbf{Interleaving}: Mix different topics during study
  \item \textbf{Productive failure}: Attempt problems before instruction to improve hippocampal efficiency
  \item \textbf{Punitive failure}: Avoid punishment-based learning
\end{itemize}
